//@version=5
indicator("Sample AL Signal", overlay=true)

fastEma = ta.ema(close, 9)
slowEma = ta.ema(close, 21)
rsiVal = ta.rsi(close, 14)

al = ta.crossover(fastEma, slowEma) and rsiVal > 45 and rsiVal < 70

plotshape(al, title="AL", style=shape.triangleup, location=location.belowbar, color=color.green, size=size.small)
