// Strong High : yeni dibi yaptıran tepe (iki dip arasındaki en yüksek yer).
// Fresh       : kırılımdan sonra fiyatın geri dönmediği range parçası.
// Hedef       : aktif strong kırılmadan önce gidilmesi beklenen weak high / weak low.
//

// Ayı bacağı : bir önceki dipten, yeni dibi getiren tepeye.

// Strong kırıldıktan sonra karşı strong da kırılırsa bu bir likidite alımıdır.

f_extreme(int fromBar, int toBar, bool wantLow) =>
    float px = na
    int pxBar = na
    int a = math.min(fromBar, toBar)
    int b = math.max(fromBar, toBar)
    int enEski = math.max(bar_index - maxGeri, 0)
    if b > bar_index
        b := bar_index
    if a < enEski
        a := enEski
    if b >= a and a >= 0
        int offB = bar_index - b
        px := wantLow ? low[offB] : high[offB]
        pxBar := b
        int span = math.min(b - a, 480)
        if span > 0
            for step = 1 to span
                int barX = b - step
                int off = bar_index - barX
                float v = wantLow ? low[off] : high[off]
                if wantLow ? v < px : v > px
                    px := v
                    pxBar := barX
    [px, pxBar]

f_breakBar(int fromBar, int toBar, float level, int dir) =>
    int found = na
    int a = math.max(fromBar + 1, bar_index - maxGeri)
    int b = math.min(toBar, bar_index)
    int span = math.min(b - a, 480)
    if span >= 0 and not na(level)
        for step = 0 to span
            int barX = a + step
            int off = bar_index - barX
            bool kirildi = dir == 1 ? (kirilimKapanis ? close[off] > level : high[off] > level) : (kirilimKapanis ? close[off] < level : low[off] < level)
            if kirildi
                found := barX
                break
    found

f_bigBreak(int fromBar, int toBar, float level, int dir) =>
    bool big = false
    int a = math.max(fromBar, bar_index - maxGeri)
    int b = math.min(toBar, bar_index)
    int span = math.min(b - a, 480)
    if span >= 0 and not na(level)
        for step = 0 to span
            int off = bar_index - (a + step)
            float govde = math.abs(close[off] - open[off])
            bool kirildi = dir == 1 ? close[off] > level : close[off] < level
            if kirildi and govde >= atr[off] * dispMult
                big := true
                break
    big

f_applyVisit(FreshZone z, float lo, float hi) =>
    if z.dir == 1
        if lo <= z.freshBot
            z.alive := false
        else if lo < z.freshTop
            z.freshTop := lo
    else
        if hi >= z.freshTop
            z.alive := false
        else if hi > z.freshBot
            z.freshBot := hi
    if z.alive and z.freshTop - z.freshBot <= syminfo.mintick * 2
        z.alive := false
    z.alive

f_zoneText(FreshZone z, bool aktif) =>
    string t = z.liq and z.big ? "Fresh · likidite · sert" : z.liq ? "Fresh · likidite" : z.big ? "Fresh · sert mum" : "Fresh"
    aktif ? t + " · aktif" : t

f_paintZone(FreshZone z, bool aktif) =>
    color govde = z.dir == 1 ? color.new(#1F8A70, 88) : color.new(#C44536, 88)
    color cizgi = z.liq ? color.new(#C6A15B, 0) : z.dir == 1 ? color.new(#0E7C66, 0) : color.new(#A33B32, 0)
    if not na(z.bx)
        box.set_top(z.bx, z.freshTop)
        box.set_bottom(z.bx, z.freshBot)
        box.set_right(z.bx, bar_index + 1)
        box.set_bgcolor(z.bx, govde)
        box.set_border_color(z.bx, cizgi)
        if aktif
            box.set_border_width(z.bx, 2)
        else
            box.set_border_width(z.bx, 1)
        box.set_text(z.bx, f_zoneText(z, aktif))

f_pushZone(bool yap, float top, float bot, int leftBar, int breakBar, int dir, bool liq, bool buyuk) =>
    float zTop = math.max(top, bot)
    float zBot = math.min(top, bot)
    if yap and zTop > zBot and not na(breakBar)
        bool benzer = false
        if array.size(zones) > 0
            FreshZone son = array.get(zones, array.size(zones) - 1)
            float birlesim = math.max(son.top, zTop) - math.min(son.bot, zBot)
            float kesisim = math.min(son.top, zTop) - math.max(son.bot, zBot)
            if son.alive and son.dir == dir and birlesim > 0 and kesisim / birlesim >= 0.7
                benzer := true
                if liq
                    son.liq := true
        if not benzer and showFresh
            int bas = math.max(leftBar, bar_index - maxGeri)
            FreshZone z = FreshZone.new(zTop, zBot, zTop, zBot, bas, breakBar, dir, buyuk, liq, true, na)
            int sag = math.min(bar_index, breakBar + 480)
            int span = math.min(sag - breakBar, 480)
            if span > 0
                for step = 1 to span
                    int off = bar_index - (breakBar + step)
                    if not f_applyVisit(z, low[off], high[off])
                        break
            if z.alive
                color govde = dir == 1 ? color.new(#1F8A70, 88) : color.new(#C44536, 88)
                color cizgi = liq ? color.new(#C6A15B, 0) : dir == 1 ? color.new(#0E7C66, 0) : color.new(#A33B32, 0)
                z.bx := box.new(bas, z.freshTop, bar_index + 1, z.freshBot, xloc=xloc.bar_index, bgcolor=govde, border_color=cizgi, border_width=1, text=f_zoneText(z, false), text_size=size.tiny, text_color=chart.fg_color, text_halign=text.align_right, text_valign=text.align_center)
                array.push(zones, z)
                while array.size(zones) > maxZone
                    FreshZone eski = array.shift(zones)
                    if not na(eski.bx)
                        box.delete(eski.bx)

f_liqYaz(string metin, float y, bool yukari) =>
    if showLiq
        label lb = label.new(bar_index, y, metin, style=yukari ? label.style_label_down : label.style_label_up, color=color.new(#C6A15B, 0), textcolor=color.new(#1B1408, 0), size=size.tiny)
        array.push(liqLabels, lb)
        while array.size(liqLabels) > 12
            label eski = array.shift(liqLabels)
            label.delete(eski)

f_setLine(line ln, int origin, float y, color col, bool dashed) =>
    line sonuc = ln
    if na(y)
        if not na(ln)
            line.delete(ln)
        sonuc := na
    else
        int x1 = na(origin) ? bar_index : origin
        if na(ln)
            if dashed
                sonuc := line.new(x1, y, bar_index, y, xloc=xloc.bar_index, extend=extend.right, color=col, width=2, style=line.style_dashed)
            else
                sonuc := line.new(x1, y, bar_index, y, xloc=xloc.bar_index, extend=extend.right, color=col, width=2, style=line.style_solid)
        else
            line.set_xy1(ln, x1, y)
            line.set_xy2(ln, bar_index, y)
            line.set_color(ln, col)
            if dashed
                line.set_style(ln, line.style_dashed)
            else
                line.set_style(ln, line.style_solid)
            sonuc := ln
    sonuc

f_setLabel(label lb, float y, string txt, color col) =>
    label sonuc = lb
    if na(y) or txt == ""
        if not na(lb)
            label.delete(lb)
        sonuc := na
    else
        if na(lb)
            sonuc := label.new(bar_index, y, txt, xloc=xloc.bar_index, style=label.style_label_left, color=col, textcolor=color.white, size=size.small)
        else
            label.set_xy(lb, bar_index, y)
            label.set_text(lb, txt)
            label.set_color(lb, col)
            sonuc := lb
    sonuc
